import { createTheme } from "@mui/material";

const getCustomTheme = (mode) =>
  createTheme({
    palette: {
      mode, // "light" ya "dark" dynamic hoga
      primary: {
        main: "#00927c",
      },
      secondary: {
        main: "#EAF0F1",
      },
      ...(mode === "dark" && {
        background: {
          default: "#121212",
          paper: "#1e1e1e",
        },
        text: {
          primary: "#ffffff",
          secondary: "#cccccc",
        },
      }),
    },
  });

export default getCustomTheme;

use:
<ThemeProvider theme={getCustomTheme("light")}>...</ThemeProvider>

# App.tsx

import { ThemeProvider } from "@mui/material";
import { RouterProvider, createBrowserRouter } from "react-router-dom";
import customTheme from "./Theme/customTheme";

import { customerRoutes } from "./routes/customerRoutes";
import { sellerRoutes } from "./routes/SellerRoutes";
import BecomeSeller from "./Auth/Become Seller/BecomeSeller";
import AdminDashboard from "./admin/Dashboard/AdminDashboard";

// Merge all routes
const appRoutes = createBrowserRouter([
  ...customerRoutes,
  ...sellerRoutes,
  {
    path: "/become-seller",
    element: <BecomeSeller />,
  },
  {
    path: "/admin/*",
    element: <AdminDashboard />,
  },
  {
    path: "*",
    element: <h2 className="text-center mt-20">404 - Page Not Found</h2>,
  },
]);

function App() {
  return (
    <ThemeProvider theme={customTheme}>
      <RouterProvider router={appRoutes} />
    </ThemeProvider>
  );
}

export default App;
